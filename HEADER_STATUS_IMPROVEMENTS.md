## 🔧 Code Implementation

### HTML Structure:
```html
<div id="api-quick-status" class="flex items-center gap-2">
    <!-- Generated by JavaScript -->
</div>
```

### JavaScript (Enhanced):
```javascript
function updateQuickStatusWithActivity() {
    const quickStatus = document.getElementById('api-quick-status');
    const keys = window.apiKeyManager.keys;
    
    const providers = [
        { id: 'gemini', icon: '🔷', letter: 'G', short: 'Gem', name: 'Gemini' },
        { id: 'openai', icon: '🟢', letter: 'O', short: 'OAI', name: 'OpenAI' },
        { id: 'anthropic', icon: '🟣', letter: 'C', short: 'Ant', name: 'Claude' },
        { id: 'perplexity', icon: '🔵', letter: 'P', short: 'Ppl', name: 'Perplexity' },
        { id: 'deepseek', icon: '💠', letter: 'D', short: 'DS', name: 'DeepSeek' }
    ];
    
    quickStatus.innerHTML = providers.map(p => {
        const hasKey = keys[p.id];
        const isAnalyzing = window.activeAPIs.analyzing.has(p.id);
        const wasUsed = window.activeAPIs.used.has(p.id);
        
        let statusIcon, statusText, className;
        
        if (isAnalyzing) {
            statusIcon = '⚡';
            statusText = 'Analyzing';
            className = 'opacity-100 animate-pulse';
        } else if (wasUsed) {
            statusIcon = '✅';
            statusText = 'Just Used';
            className = 'opacity-100';
        } else if (hasKey) {
            statusIcon = '';
            statusText = 'Ready';
            className = 'opacity-100';
        } else {
            statusIcon = '';
            statusText = 'Not Set';
            className = 'opacity-40';
        }
        
        // Responsive display
        return `
            <span class="flex items-center gap-0.5 ${className} transition-all duration-200" 
                  title="${p.name}: ${statusText}">
                <span class="text-sm">${p.icon}</span>
                <span class="text-xs font-bold">${statusIcon || p.letter}</span>
                <span class="text-xs font-medium hidden sm:inline">${p.short}</span>
                <span class="text-xs font-medium hidden md:inline">${p.name}</span>
            </span>
        `;
    }).join('');
}
```

### CSS (Responsive):
